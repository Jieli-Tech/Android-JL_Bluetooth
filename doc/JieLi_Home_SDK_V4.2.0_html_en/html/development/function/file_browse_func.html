25.1.2. FileObserver

Directory browsing observer

public interface FileObserver {

     /**
      * Callback after receiving directory file data
      *
      * @param fileStructs File structure list
      */
     void onFileReceiver(List<FileStruct> fileStructs);

     /**
      * One file reading ends
      *
      * @param isEnd Whether ended
      */
     void onFileReadStop(boolean isEnd);

     /**
      * File reading starts
      */
     void onFileReadStart();

     /**
      * File reading failed
      *
      * @param reason Error code
      */
     void onFileReadFailed(int reason);

     /**
      * Storage device status change of device
      *
      * @param onLineCards Online device list
      */
     void onSdCardStatusChange(List<> onLineCards);

     /**
      * File on-demand success callback
      */
     void OnFlayCallback(boolean success);

 }

Note

  1. File structure, FileStruct

  2. Storage information, SDCardBean

25.2. Directory Browsing

FileObserver fileObserver = new FileObserver() {
    @Override
    public void onFileReceiver(List<FileStruct> fileStructs) {
        // Read file list, only callback current reading file list
    }

    @Override
    public void onFileReadStop(boolean isEnd) {
        // File list reading ends
    }

    @Override
    public void onFileReadStart() {
        // Start file list reading
    }

    @Override
    public void onFileReadFailed(int reason) {
        // File list reading failed
    }

    @Override
    public void onSdCardStatusChange(List<> onLineCards) {
        //Online device changes
    }

    @Override
    public void OnFlayCallback(boolean success) {
        //Song on-demand callback
    }
};
//Step 1: Register observer

// Step 2: Get online device list, can handle device status changes through fileObserver
FileBrowseManager.getInstance().addFileObserver(fileObserver);

// Step 3: Read current directory currently being read by current device
List<> list = FileBrowseManager.getInstance().getOnlineDev();
if(list.size()<1){
    //No online devices
    return;
}
  = list.get(0);//Get device, if multiple devices, get required device according to needs
Folder currentFolder = FileBrowseManager.getInstance().getCurrentReadFile();

//Step 4: Get sub-files already read but in cache under current directory
List<FileStruct> fileStructs = currentFolder.getChildFileStructs();

//Step 5: Browse operation
//Load more
FileBrowseManager.getInstance().loadMore();
//Enter next level directory
FileStruct fileStruct = currentFolder.getChildFileStructs().get(0);//Get required folder according to needs
FileBrowseManager.getInstance().appenBrowse(fileStruct, );
//Return to previous directory without list callback
boolean hasEvent = true;//Whether need FileObserver event callback
FileBrowseManager.getInstance().backBrowse(,hasEvent);
//On-demand file
FileBrowseManager.getInstance().playFile(fileStruct, );
