1. RCSP Basic Interface

public interface IRcspOp {


    /**
     * Register Rcsp event callback
     *
     * @param callback RCSP event callback
     */
    void registerOnRcspCallback(OnRcspCallback callback);

    /**
     * Unregister RCSP event callback
     *
     * @param callback RCSP event callback
     */
    void unregisterOnRcspCallback(OnRcspCallback callback);

    /**
     * Register RCSP event listener
     *
     * @param listener RCSP event listener
     */
    void registerOnRcspEventListener(OnRcspEventListener listener);

    /**
     * Unregister RCSP event listener
     *
     * @param listener RCSP event listener
     */
    void unregisterOnRcspEventListener(OnRcspEventListener listener);

    /**
     * Whether RCSP protocol is initialized
     * <p>
     *     Default is current using device
     * </p>
     * @return boolean Result
     */
    boolean isRcspInit();

    /**
     * Whether RCSP protocol is initialized
     *
     * @param device Bluetooth device
     * @return boolean Result
     */
    boolean isRcspInit(BluetoothDevice device);

    /**
     * Get Bluetooth library configuration
     *
     * @return Bluetooth library configuration
     */
    BluetoothOption getBluetoothOption();

    /**
     * Get device info
     *
     * @return {@link DeviceInfo} Device info
     */
    DeviceInfo getDeviceInfo();

    /**
     * Get device info
     *
     * @param device Bluetooth device
     * @return {@link DeviceInfo} Device info
     */
    DeviceInfo getDeviceInfo(BluetoothDevice device);

    /**
     * Get device configuration info
     *
     * @param device Bluetooth device
     * @return DeviceConfiguration Device configuration info
     */
    DeviceConfiguration getDeviceConfiguration(BluetoothDevice device);

    /**
     * Configure Bluetooth library parameters
     *
     * @param option Parameter configuration
     */
    void configure(BluetoothOption option);

    /**
     * Send RCSP command
     *
     * @param device    Bluetooth device object
     * @param cmd       RCSP command
     * @param timeoutMs Timeout. Unit is milliseconds
     * @param callback  Send callback
     */
    void sendCommandAsync(BluetoothDevice device, CommandBase cmd, int timeoutMs, RcspCommandCallback callback);

    /**
     * Send RCSP command
     *
     * <p>
     * Equivalent to {@link #sendCommandAsync(BluetoothDevice, CommandBase, int, RcspCommandCallback)},<br/>
     * To ensure interface consistency
     * </p>
     *
     * @param device    Bluetooth device object
     * @param cmd       RCSP command
     * @param timeoutMs Timeout. Unit is milliseconds
     * @param callback  Send callback
     */
    void sendRcspCommand(BluetoothDevice device, CommandBase cmd, int timeoutMs, RcspCommandCallback callback);

    /**
     * Send RCSP command
     *
     * @param device   Bluetooth device object
     * @param cmd      RCSP command
     * @param callback Send callback
     */
    void sendRcspCommand(BluetoothDevice device, CommandBase cmd, RcspCommandCallback callback);

    /**
     * Send RCSP response command
     *
     * @param device      Bluetooth device object
     * @param responseCmd RCSP response command
     * @param callback    Send callback
     */
    void sendCommandResponse(BluetoothDevice device, CommandBase responseCmd, RcspCommandCallback callback);

    /**
     * Send RCSP response command
     * <p>
     * Equivalent to {@link #sendCommandResponse(BluetoothDevice, CommandBase, RcspCommandCallback)},<br/>
     * To ensure interface consistency
     * </p>
     *
     * @param device Bluetooth device object
     * @param responseCmd RCSP response command
     */
    void sendRcspResponse(BluetoothDevice device, CommandBase responseCmd);

    /**
     * Switch connected device to currently using device
     * <p>
     * Can get switching status via {@link OnRcspCallback#onSwitchConnectedDevice(BluetoothDevice)}
     * </p>
     *
     * @param device Connected device
     */
    void switchConnectedDevice(BluetoothDevice device);

    /**
     * Release resources
     */
    void release();

}

    private final int type;   //Data type
    private int version;      //Version number
}
