1.1 Command reply callback

public interface RcspCommandCallback {

    /**
     * Callback reply command
     *
     * @param device Connected device
     * @param cmd  Reply command
     */
    void onCommandResponse(BluetoothDevice device, CommandBase cmd);

    /**
     * Callback error event
     *
     * @param device Connected device
     * @param error  error message
     */
    void onErrCode(BluetoothDevice device, BaseError error);
}

1.2 Bluetooth Operation Interface

public interface IBluetoothOperation {

    /*-------------------------------- BluetoothAdapter Op ------------------------*/

    /**
     * Checks whether the Bluetooth module is enabled.
     *
     * @return boolean Result. Returns true if Bluetooth is enabled; false otherwise.
     */
    boolean isBluetoothEnabled();

    /**
     * Enables the Bluetooth module.
     *
     * <p>Requires dynamic user permission.</p>
     *
     * @return boolean Operation result
     */
    boolean enableBluetooth();

    /**
     * Disables the Bluetooth module.
     *
     * @return boolean Operation result
     */
    boolean disableBluetooth();

    /**
     * Registers a Bluetooth callback listener.
     *
     * @param bluetoothCallback Bluetooth callback listener. It is recommended to use {@link BluetoothCallbackImpl}.
     * @return boolean Operation result
     */
    boolean registerBluetoothCallback(IBluetoothCallback bluetoothCallback);

    /**
     * Unregisters a Bluetooth callback listener.
     *
     * @param bluetoothCallback Bluetooth callback listener. It is recommended to use {@link BluetoothCallbackImpl}.
     * @return boolean Operation result
     */
    boolean unregisterBluetoothCallback(IBluetoothCallback bluetoothCallback);

    /**
     * Returns the currently connected Bluetooth device object.
     *
     * <p>In BLE mode, returns the BLE device object.<br/>
     * In SPP mode, note that the connection may be established via BLE first and then converted to an SPP connection,
     * so it may first return the BLE object, followed by the SPP (classic Bluetooth) object.</p>
     *
     * @return The communication channel object. Returns null if no communication channel is established.
     */
    BluetoothDevice getConnectedDevice();

    /**
     * Sets the currently connected and active Bluetooth device.
     *
     * @param device Bluetooth device
     */
    void setConnectedDevice(BluetoothDevice device);

    /**
     * Gets the currently used BluetoothGatt object.
     *
     * @return BluetoothGatt The currently used BluetoothGatt object
     */
    BluetoothGatt getConnectedBluetoothGatt();

    /**
     * Gets the BluetoothGatt object for a specified device.
     *
     * @param device BLE device
     * @return BluetoothGatt The connected BluetoothGatt object
     */
    BluetoothGatt getDeviceGatt(BluetoothDevice device);

    /**
     * Gets the list of connected devices.
     *
     * @return List<BluetoothDevice> List of connected devices
     */
    List<BluetoothDevice> getConnectedDeviceList();

    /**
     * Checks whether a device is connected.
     *
     * @param device Bluetooth device
     * @return boolean Result
     */
    boolean isConnectedDevice(BluetoothDevice device);

    /**
     * Sets Bluetooth configuration options.
     *
     * @param bluetoothOption Bluetooth configuration options
     */
    void setBluetoothOption(BluetoothOption bluetoothOption);

    /**
     * Gets Bluetooth configuration options.
     *
     * @return BluetoothOption Bluetooth configuration options
     */
    BluetoothOption getBluetoothOption();

    /**
     * Gets a remote Bluetooth device object.
     *
     * @param address Bluetooth device address
     * @return BluetoothDevice Bluetooth device object. Returns the device object if it exists; otherwise, returns null.
     */
    BluetoothDevice getRemoteDevice(String address);

    /**
     * Gets the cached classic Bluetooth device object.
     *
     * @param device Bluetooth device
     * @return BluetoothDevice Cached classic Bluetooth device object
     */
    BluetoothDevice getCacheEdrDevice(BluetoothDevice device);

    /*-----------------------------------scan device op--------------------------*/

    /**
     * Checks whether device scanning is in progress.
     *
     * @return boolean Result
     */
    boolean isScanning();

    /**
     * Gets the list of discovered Bluetooth devices.
     *
     * <p>The list only includes devices of the scanned type. Scanning operations clear the cached list.</p>
     *
     * @return ArrayList<BluetoothDevice> List of discovered Bluetooth devices
     */
    ArrayList<BluetoothDevice> getDiscoveredBluetoothDevices();

    /**
     * Gets the current scan type.
     *
     * @return int Scan type
     * <p>
     * {@link com.jieli.bluetooth.constant.BluetoothConstant#SCAN_TYPE_BLE} --- BLE devices <br/>
     * {@link com.jieli.bluetooth.constant.BluetoothConstant#SCAN_TYPE_CLASSIC} --- Classic Bluetooth devices
     * </p>
     */
    int getScanType();

    /**
     * Starts scanning for Bluetooth devices.
     *
     * <p>This method scans all Bluetooth devices, but the SDK only reports classic Bluetooth devices via callbacks.</p>
     * <p>If no device is found before the timeout, a scan timeout error is reported via {@link IBluetoothCallback#onError(BaseError)}.</p>
     *
     * @param timeout Timeout value (in milliseconds)
     * @return int Operation result code
     * <p>
     * {@link com.jieli.bluetooth.constant.ErrorCode#ERR_NONE} --- Success<br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed<br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed<br/>
     * <br/>
     * If successful, scan status will be reported via {@link IBluetoothCallback#onDiscoveryStatus(boolean, boolean)}.<br/>
     * Discovered devices will be reported via {@link IBluetoothCallback#onDiscovery(BluetoothDevice, BleScanMessage)}.<br/>
     * Pop-up products will trigger {@link IBluetoothCallback#onShowDialog(BluetoothDevice, BleScanMessage)}.
     * </p>
     */
    int startDeviceScan(long timeout);

    /**
     * Stops scanning for Bluetooth devices.
     *
     * @return int Operation result code
     * <p>
     * {@link com.jieli.bluetooth.constant.ErrorCode#ERR_NONE} --- Success<br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed<br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed
     * </p>
     */
    int stopDeviceScan();

    /**
     * Starts scanning for BLE devices.
     *
     * <p>If no device is found before the timeout, a scan timeout error is reported via {@link IBluetoothCallback#onError(BaseError)}.</p>
     *
     * @param timeout Timeout value (in milliseconds)
     * @return int Operation result code
     * <p>
     * {@link ErrorCode#ERR_NONE} --- Success<br/>
     * {@link ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed<br/>
     * {@link ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed <br/>
     * <br/>
     * If successful, scan status will be reported via {@link IBluetoothCallback#onDiscoveryStatus(boolean, boolean)}.<br/>
     * Discovered devices will be reported via {@link IBluetoothCallback#onDiscovery(BluetoothDevice, BleScanMessage)}.<br/>
     * Pop-up products will trigger {@link IBluetoothCallback#onShowDialog(BluetoothDevice, BleScanMessage)}.
     * </p>
     */
    int startBLEScan(long timeout);

    /**
     * Stops scanning for BLE devices.
     *
     * @return int Operation result code
     * <p>
     * {@link ErrorCode#ERR_NONE} --- Success<br/>
     * {@link ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed
     * </p>
     */
    int stopBLEScan();

    /*-----------------------------------pair op--------------------------*/

    /**
     * Checks whether a specific device is currently pairing.
     *
     * @param device Bluetooth device
     * @return boolean Result
     */
    boolean isPairing(BluetoothDevice device);

    /**
     * Checks whether a specific device is already paired.
     *
     * @param device Bluetooth device
     * @return boolean Result
     */
    boolean isPaired(BluetoothDevice device);

    /**
     * Gets the list of paired devices.
     *
     * @return List<BluetoothDevice> List of paired devices
     */
    List<BluetoothDevice> getPairedDevices();

    /**
     * Starts the pairing process for a specified device.
     *
     * @param device Bluetooth device
     * @return int Result code
     * <p>
     * {@link ErrorCode#ERR_NONE} --- Success <br/>
     * {@link ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed <br/>
     * {@link ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed<br/>
     * If successful, status will be reported via {@link OnBtDevicePairListener#onBtDeviceBond(BluetoothDevice, int)}.
     * </p>
     */
    int pair(BluetoothDevice device);

    /**
     * Starts the pairing process for a specified device.
     *
     * @param device  Bluetooth device
     * @param pairWay Pairing method, valid range: [0, 2]
     * @return int Result code
     * <p>
     * {@link ErrorCode#ERR_NONE} --- Success <br/>
     * {@link ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed <br/>
     * {@link ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed<br/>
     * If successful, status will be reported via {@link OnBtDevicePairListener#onBtDeviceBond(BluetoothDevice, int)}.
     * </p>
     */
    int pair(BluetoothDevice device, @IntRange(from = 0, to = 2) int pairWay);

    /**
     * Cancels pairing with a Bluetooth device.
     *
     * @param device Bluetooth device
     * @return int Operation code
     * <p>
     * {@link ErrorCode#ERR_NONE} --- Success <br/>
     * {@link ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed <br/>
     * {@link ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed<br/>
     * If successful, status will be reported via {@link OnBtDevicePairListener#onBtDeviceBond(BluetoothDevice, int)}.
     * </p>
     */
    int unPair(BluetoothDevice device);

    /**
     * Attempts to start the pairing process for a specified device.
     *
     * <p>Performs pairing in a queued manner to avoid excessive operations that could destabilize the Bluetooth system. This method is recommended for pairing.</p>
     *
     * @param device Bluetooth device
     * @return boolean Operation result
     */
    boolean tryToPair(BluetoothDevice device);

    /**
     * Attempts to start the pairing process for a specified device.
     *
     * <p>Performs pairing in a queued manner to avoid excessive operations that could destabilize the Bluetooth system. This method is recommended for pairing.</p>
     *
     * @param device  Bluetooth device
     * @param pairWay Pairing method, valid range: [0, 2]
     * @return boolean Operation result
     */
    boolean tryToPair(BluetoothDevice device, @IntRange(from = 0, to = 2) int pairWay);

    /**
     * Attempts to start the unpairing process for a specified device.
     *
     * <p>Performs unpairing in a queued manner to avoid excessive operations that could destabilize the Bluetooth system. This method is recommended for unpairing.</p>
     *
     * @param device Bluetooth device
     * @return boolean Operation result
     */
    boolean tryToUnPair(BluetoothDevice device);

    /*-----------------------------------br/edr op--------------------------*/

    /**
     * Checks whether a Bluetooth device supports the HFP profile.
     *
     * @param device Classic Bluetooth device
     * @return boolean Result
     */
    boolean deviceHasHfp(BluetoothDevice device);

    /**
     * Checks whether a Bluetooth device supports the A2DP profile.
     *
     * @param device Classic Bluetooth device
     * @return boolean Result
     */
    boolean deviceHasA2dp(BluetoothDevice device);

    /**
     * Checks whether an HFP connection has been established with a Bluetooth device.
     *
     * @param device Classic Bluetooth device
     * @return int Connection status<p>
     * {@link android.bluetooth.BluetoothProfile#STATE_DISCONNECTED} --- Disconnected <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_CONNECTING} --- Connecting <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_CONNECTED} --- Connected <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_DISCONNECTING} --- Disconnecting
     * </p>
     */
    int isConnectedByHfp(BluetoothDevice device);

    /**
     * Checks whether an A2DP connection has been established with a Bluetooth device.
     *
     * @param device Classic Bluetooth device
     * @return int Connection status<p>
     * {@link android.bluetooth.BluetoothProfile#STATE_DISCONNECTED} --- Disconnected <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_CONNECTING} --- Connecting <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_CONNECTED} --- Connected <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_DISCONNECTING} --- Disconnecting
     * </p>
     */
    int isConnectedByA2dp(BluetoothDevice device);

    /**
     * Checks whether a classic Bluetooth profile (A2DP or HFP) connection has been established with a device.
     *
     * @param device Classic Bluetooth device
     * @return int Connection status<p>
     * {@link android.bluetooth.BluetoothProfile#STATE_DISCONNECTED} --- Disconnected <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_CONNECTING} --- Connecting <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_CONNECTED} --- Connected <br/>
     * {@link android.bluetooth.BluetoothProfile#STATE_DISCONNECTING} --- Disconnecting
     * </p>
     */
    int isConnectedByProfile(BluetoothDevice device);

    /**
     * Checks whether a classic Bluetooth connection is in progress.
     *
     * @return boolean Result
     */
    boolean isBrEdrConnecting();

    /**
     * Gets the classic Bluetooth device currently being connected.
     *
     * @return BluetoothDevice Classic Bluetooth device
     */
    BluetoothDevice getConnectingBrEdrDevice();

    /**
     * Connects via classic Bluetooth profiles.
     *
     * @param device Classic Bluetooth device
     * @return boolean Operation result <p>
     * If successful, connection status will be reported via {@link IBluetoothCallback#onBrEdrConnection(BluetoothDevice, int)}.
     * </p>
     */
    boolean connectByProfiles(BluetoothDevice device);

    /**
     * Disconnects a classic Bluetooth connection.
     *
     * @param device Classic Bluetooth device
     * @return boolean Result
     */
    boolean disconnectByProfiles(BluetoothDevice device);

    /**
     * Sets the specified connected device as the active audio output device.
     *
     * @param device Specified connected device
     * @return boolean Result
     */
    boolean setActivityBluetoothDevice(BluetoothDevice device);

    /**
     * Gets the current active audio output Bluetooth device.
     *
     * @return BluetoothDevice Current active audio output Bluetooth device
     */
    BluetoothDevice getActivityBluetoothDevice();

    /**
     * Gets the list of connected SPP devices.
     *
     * @return List<BluetoothDevice> List of connected SPP devices
     */
    List<BluetoothDevice> getConnectedSppList();

    /**
     * Checks whether a device is connected via SPP.
     *
     * @param device Bluetooth device
     * @return boolean Result
     */
    boolean isConnectedSppDevice(BluetoothDevice device);

    /**
     * Connects an SPP channel.
     *
     * @param device Classic Bluetooth device
     * @return int Operation result
     * <p>
     * {@link ErrorCode#ERR_NONE} --- Success <br/>
     * {@link ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed <br/>
     * {@link ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed<br/>
     * If successful, connection status will be reported via {@link IBluetoothCallback#onSppStatus(BluetoothDevice, int)}.
     * </p>
     */
    int connectSPPDevice(BluetoothDevice device);

    /**
     * Disconnects an SPP channel.
     *
     * @param device Classic Bluetooth device
     * @return boolean Operation result
     * <p>
     * If successful, connection status will be reported via {@link IBluetoothCallback#onSppStatus(BluetoothDevice, int)}.
     * </p>
     */
    boolean disconnectSPPDevice(BluetoothDevice device);

    /**
     * Sends a data packet via the SPP channel.
     *
     * @param device Classic Bluetooth device
     * @param data   Data packet
     * @return boolean Operation result
     */
    boolean writeDataToSppDevice(BluetoothDevice device, byte[] data);

    /**
     * Gets the list of connected classic Bluetooth devices.
     *
     * @return List<BluetoothDevice> List of connected classic Bluetooth devices
     */
    List<BluetoothDevice> getDevicesConnectedByProfile();

    /*----------------------------------ble op----------------------------*/

    /**
     * Gets the list of paired BLE devices.
     *
     * @return List<BluetoothDevice> List of paired BLE devices, or null if none exist

    List<BluetoothDevice> getPairedBLEDevices();

    /**
     * Gets the negotiated MTU of a connected BLE device.

     * @param device Connected BLE device

    int getBleMtu(BluetoothDevice device);

    /**

    /**
     * Enables or disables BLE characteristic notifications.
     *

    boolean enableBLEDeviceNotification(BluetoothGatt gatt, BleCharacteristic bleCharacteristic);

    /**
     * Connects to a specified BLE device.
     *
     * @param device Bluetooth device
     * @return int Operation code<p>
     * {@link com.jieli.bluetooth.constant.ErrorCode#ERR_NONE} --- Success<br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed <br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_BLE_CONNECTING} --- BLE connecting<br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_BLE_CONNECT_FAILED} --- BLE connection failed<br/>
     * <br/>
     * If successful, connection status will be reported via {@link OnBtBleListener#onBleConnection(BluetoothDevice, int)}.
     * </p>
     */
    int connectBLEDevice(BluetoothDevice device);

    /**
     * Connects to a specified BLE device.
     *
     * @param device Bluetooth device
     * @param param  Connection parameters
     * @return int Operation code<p>
     * {@link com.jieli.bluetooth.constant.ErrorCode#ERR_NONE} --- Success<br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed <br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_BLE_CONNECTING} --- BLE connecting<br/>
     * {@link com.jieli.bluetooth.constant.ErrorCode#SUB_ERR_BLE_CONNECT_FAILED} --- BLE connection failed<br/>
     * <p>
     * If successful, connection status will be reported via {@link OnBtBleListener#onBleConnection(BluetoothDevice, int)}.
     * </p>
     */
    int connectBLEDevice(BluetoothDevice device, @NonNull BleConnectParam param);

    /**
     * Disconnects a BLE device.
     *
     * @param device BLE device
     * @return boolean Operation result<p>
     * If successful, connection status will be reported via {@link IBluetoothCallback#onBleConnection(BluetoothDevice, int)}.
     * </p>
     */
    boolean disconnectBLEDevice(BluetoothDevice device);

    /**
     * Sends a data packet via BLE write service.
     *
     * @param device             BLE device
     * @param serviceUUID        Service UUID
     * @param characteristicUUID Characteristic UUID
     * @param writeData          Data packet
     * @return boolean Send result
     */
    boolean writeDataToBLEDevice(BluetoothDevice device, UUID serviceUUID, UUID characteristicUUID, byte[] writeData);

    /*---------------------------------- special op ----------------------------*/

    /**
     * Synchronizes connection history records.
     */

    /**
     * Gets the device currently being connected.
     *
     * @return BluetoothDevice Device currently being connected

    BluetoothDevice getConnectingDevice();

    /**
     * Connects a Bluetooth device.
     *
     * <p>Connects based on device type: BLE devices use BLE connection; classic Bluetooth devices use classic Bluetooth connection.</p>
     *
     * @param device Bluetooth device
     * @return int Result code<p>
     * {@link ErrorCode#ERR_NONE} --- Success<br/>
     * {@link ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed <br/>
     * {@link ErrorCode#SUB_ERR_BLE_CONNECTING} --- BLE connecting<br/>
     * {@link ErrorCode#SUB_ERR_BLE_CONNECT_FAILED} --- BLE connection failed<br/>
     * {@link ErrorCode#SUB_ERR_SPP_CONNECTING} --- SPP connecting<br/>
     * {@link ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed <br/>
     * {@link ErrorCode#SUB_ERR_DEVICE_CONNECTING}  --- Device connecting <br/>
     * {@link ErrorCode#SUB_ERR_DEVICE_IS_CONNECTED} --- Device already connected<br/>
     * <br/>
     * If successful, connection status will be reported via {@link IBluetoothCallback#onConnection(BluetoothDevice, int)}.
     * </p>
     */
    int connectBtDevice(BluetoothDevice device);

    /**
     * Connects a Bluetooth device.
     *
     * <p>Connects based on device type: BLE devices use BLE connection; classic Bluetooth devices use classic Bluetooth connection.</p>
     *
     * @param device     Bluetooth device
     * @param connectWay Connection method
     *                   <p>
     *                   {@link com.jieli.bluetooth.constant.BluetoothConstant#PROTOCOL_TYPE_BLE} --- BLE mode <br/>
     *                   {@link com.jieli.bluetooth.constant.BluetoothConstant#PROTOCOL_TYPE_SPP} --- SPP mode<br/>
     *                   {@link com.jieli.bluetooth.constant.BluetoothConstant#PROTOCOL_TYPE_GATT_OVER_BR_EDR} -- GATT over BR/EDR mode
     *                   </p>
     * @return int Result code<p>
     * {@link ErrorCode#ERR_NONE} --- Success<br/>
     * {@link ErrorCode#SUB_ERR_OP_FAILED} --- Operation failed <br/>
     * {@link ErrorCode#SUB_ERR_BLE_CONNECTING} --- BLE connecting<br/>
     * {@link ErrorCode#SUB_ERR_BLE_CONNECT_FAILED} --- BLE connection failed<br/>
     * {@link ErrorCode#SUB_ERR_SPP_CONNECTING} --- SPP connecting<br/>
     * {@link ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed <br/>
     * {@link ErrorCode#SUB_ERR_DEVICE_CONNECTING}  --- Device connecting <br/>
     * {@link ErrorCode#SUB_ERR_DEVICE_IS_CONNECTED} --- Device already connected<br/>
     * <br/>
     * If successful, connection status will be reported via {@link IBluetoothCallback#onConnection(BluetoothDevice, int)}.
     * </p>
     */
    int connectBtDevice(BluetoothDevice device, int connectWay);

    /**
     * Disconnects a Bluetooth device.
     *
     * <p>Connection status will be reported via {@link IBluetoothCallback#onConnection(BluetoothDevice, int)}.</p>
     *
     * @param device Bluetooth device
     */
    void disconnectBtDevice(BluetoothDevice device);

    /**
     * Connects a classic Bluetooth device.
     *
     * @param breDevice Bluetooth device
     * @return int Result code <p>
     * {@link ErrorCode#ERR_NONE} --- Success <br/>
     * {@link ErrorCode#SUB_ERR_PARAMETER} --- Invalid parameter <br/>
     * {@link ErrorCode#SUB_ERR_EDR_CONNECTING} --- Classic Bluetooth connecting <br/>
     * {@link ErrorCode#SUB_ERR_REFLECT_WAY} --- Reflection interface call failed<br/>
     * <br/>
     * If successful, connection status will be reported via {@link IBluetoothCallback#onBtDeviceConnectStatus(BluetoothDevice, int)}.
     * </p>
     */
    int startConnectByBreProfiles(BluetoothDevice breDevice);

    /**
     * Checks whether a device connection is in progress.
     *
     * @return boolean Result
     */
    boolean isConnecting();

    /**
     * Checks whether a Bluetooth device has passed certification.
     *
     * @param device Bluetooth device
     * @return boolean Result
     */
    boolean checkDeviceIsCertify(BluetoothDevice device);

    /**
     * Fast reconnect.
     *
     * <p>Connection results are reported via {@link IBluetoothCallback}. You must register the callback using {@link #registerBluetoothCallback(IBluetoothCallback)}.</p>
     *
     * @return false if no cached BLE address exists and reconnection is not possible; true if reconnection has started.
     */
    boolean fastConnect();

    /**
     * Checks whether a BLE device is connected.
     *
     * <p>Synchronizes the system's BLE connection list to verify whether the connected BLE device is recognized by this solution.</p>
     */
    void checkBleIsConnected();

    /**
     * Synchronizes Bluetooth device status.
     *
     * <p>Verifies whether Bluetooth devices are properly connected.</p>
     */
    void synchronizationBtDeviceStatus(); // Synchronize Bluetooth device status

    /**

     * @param address Device address

    HistoryBluetoothDevice findHistoryBluetoothDevice(String address);

    /**

     */

    /**
     * Checks whether a reconnection is in progress.
     *
     * @return boolean Result
     */
    boolean isReconnecting();

    /**
     * Reconnects to a historical device record.
     *
     * @param historyBtDevice Historical device record
     * @param timeoutMs       Timeout (in milliseconds)
     * @param listener        Reconnection listener
     */
    void reconnectHistoryBluetoothDevice(HistoryBluetoothDevice historyBtDevice, long timeoutMs, OnReconnectHistoryRecordListener listener);

    /**
     * Stops reconnecting to a device.
     * @return boolean Operation result
     */
    boolean stopReconnect();

    /**
     * Releases resources.
     */
    void destroy();
