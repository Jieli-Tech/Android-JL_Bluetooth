.. _get_device_info:

Request device information
====================================================


1. JL_BluetoothManager Way

.. code-block:: java

    //Step0: Get the JL_BluetoothManager object
    JL_BluetoothManager manager = JL_BluetoothManager.getInstance(context);
    //Step1: Construct function commands - for example, get device information
    //mask = 0xffffffff;  -- Get all attributes
    CommandBase getDeviceInfoCmd = CommandBuilder.buildGetDeviceInfoCmd(mask);
    //Step2: Execute the operation command and wait for the result callback.
    manager.sendCommandAsync(manager.getConnectedDevice(), getDeviceInfoCmd, manager.getBluetoothOption().getTimeoutMs(), new RcspCommandCallback() {
        //Callback reply command
        //Note: When there is no reply command, the reply command itself
        @Override
        public void onCommandResponse(BluetoothDevice device, CommandBase cmd) {
            //Step3: Check device status
            if (cmd.getStatus() != StateCode.STATUS_SUCCESS) { //The response status is abnormal; exception handling is required.
                onErrCode(device, new BaseError(ErrorCode.SUB_ERR_RESPONSE_BAD_STATUS, "Device reply an bad status: " + cmd.getStatus()));
                return;
            }
            //Callback operation successful
            //Step4: Determine if there is any reply data
            //Note - If the command does not provide a response, the response data will be null.
            boolean isHasResponse = cmd.getType() == CommandBase.FLAG_HAVE_PARAMETER_AND_RESPONSE
                    || cmd.getType() == CommandBase.FLAG_NO_PARAMETER_AND_RESPONSE;
            if (isHasResponse) {
                //Step5: Retrieve the corresponding command data
                GetTargetInfoCmd command = (GetTargetInfoCmd) cmd;
                //Step6: Get reply data
                TargetInfoResponse response = command.getResponse();
                if (null == response) { //An empty response indicates a problem with the data returned by the device, requiring exception handling.
                    onErrCode(device, new BaseError(ErrorCode.SUB_ERR_DATA_FORMAT, "Response data is error."));
                    return;
                }
                //Processing equipment reply data
            }
        }

        @Override
        public void onErrCode(BluetoothDevice device, BaseError error) {
            //Callback operation failed
            //error - error information
        }
    });

Query device system information
===================================================================

1. JL_BluetoothManager Way

.. code-block:: java

    //Step0: Get the JL_BluetoothManager object
    JL_BluetoothManager manager = JL_BluetoothManager.getInstance(context);
    //Step1: Construct a command to query device system information
    //Function code 
    int function = AttrAndFunCode.SYS_INFO_FUNCTION_PUBLIC; //Public attribute
    //Query current device mode and device storage information
    int mask = 0x01 << AttrAndFunCode.SYS_INFO_ATTR_CUR_MODE_TYPE | 0x01 << AttrAndFunCode.SYS_INFO_ATTR_MUSIC_DEV_STATUS;
    CommandBase getSysInfoCmd = CommandBuilder.buildGetSysInfoCmd((byte)function, mask);
    //Step2: Execute the operation command and wait for the result callback.
    manager.sendCommandAsync(manager.getConnectedDevice(), getSysInfoCmd, manager.getBluetoothOption().getTimeoutMs(), new RcspCommandCallback() {
        //Callback reply command
        //Note: When there is no reply command, the reply command itself.
        @Override
        public void onCommandResponse(BluetoothDevice device, CommandBase cmd) {
            //Step3: Check device status
            if (cmd.getStatus() != StateCode.STATUS_SUCCESS) { //The response status is abnormal; exception handling is required.
                onErrCode(device, new BaseError(ErrorCode.SUB_ERR_RESPONSE_BAD_STATUS, "Device reply an bad status: " + cmd.getStatus()));
                return;
            }
            //Callback operation successful
            //Step4: Determine if there is any reply data
            //Note - If the command does not provide a response, the response data will be null.
            boolean isHasResponse = cmd.getType() == CommandBase.FLAG_HAVE_PARAMETER_AND_RESPONSE
                    || cmd.getType() == CommandBase.FLAG_NO_PARAMETER_AND_RESPONSE;
            if (isHasResponse) {
                //Step5: Retrieve the corresponding command data
                GetSysInfoCmd command = (GetSysInfoCmd) cmd;
                //Step6: Get reply data
                SysInfoResponse response = command.getResponse();
                if (null == response) { //An empty response indicates a problem with the data returned by the device, requiring exception handling.
                    onErrCode(device, new BaseError(ErrorCode.SUB_ERR_DATA_FORMAT, "Response data is error."));
                    return;
                }
                //Processing equipment reply data
                //response.getFunction(); //function code
                //response.getAttrs();    //Functional attribute
            }
        }

        @Override
        public void onErrCode(BluetoothDevice device, BaseError error) {
            //Callback operation failed
            //error - error information
        }
    });

    //Step0: Get the JL_BluetoothManager object
    JL_BluetoothManager manager = JL_BluetoothManager.getInstance(context);

    int function = AttrAndFunCode.SYS_INFO_FUNCTION_PUBLIC; //Public attribute

.. _notify_sys_info:

Monitoring device system attribute changes
=======================================================================


1. JL_BluetoothManager Way

.. code-block:: java

    //Step0: Get the JL_BluetoothManager object
    JL_BluetoothManager manager = JL_BluetoothManager.getInstance(context);
    //Step1: Add Bluetooth RCSP event callback
    manager.addEventListener(new BTRcspEventCallback() {
        //Callback device sent command
        @Override
        public void onDeviceCommand(BluetoothDevice device, CommandBase cmd) {
            super.onDeviceCommand(device, cmd);
            //Step2: Processing command
            if (cmd.getId() != Command.CMD_SYS_INFO_AUTO_UPDATE) {
                //Filtering command
                return;
            }
            //Process only target commands - notify device system information
            UpdateSysInfoCmd updateSysInfoCmd = (UpdateSysInfoCmd) cmd;
            //Step4: Get parameters
            UpdateSysInfoParam param = updateSysInfoCmd.getParam();
            if (null == param) return; //Exception handling

.. _set_function: 

Settings Function
===================================

1. JL_BluetoothManager Way

.. code-block:: java

    //Step0: Get the JL_BluetoothManager object
    JL_BluetoothManager manager = JL_BluetoothManager.getInstance(context);
    //Step1: Construct settings function command
    //Function code
    int function = AttrAndFunCode.SYS_INFO_FUNCTION_PUBLIC; //Public attribute
    //flag
    int flag = 0;
    // Extended parameters
    byte[] extend = null;
    CommandBase setFunctionCmd = CommandBuilder.buildFunctionCmd((byte) function, (byte) flag, extend);
    //Step2: Execute the operation command and wait for the result callback.
    manager.sendCommandAsync(manager.getConnectedDevice(), setFunctionCmd, manager.getBluetoothOption().getTimeoutMs(), new RcspCommandCallback() {
        //Callback reply command
        //Note: When there is no reply command, the reply command itself.
        @Override
        public void onCommandResponse(BluetoothDevice device, CommandBase cmd) {
            //Step3: Check device status
            if (cmd.getStatus() != StateCode.STATUS_SUCCESS) { //The response status is abnormal; exception handling is required.
                onErrCode(device, new BaseError(ErrorCode.SUB_ERR_RESPONSE_BAD_STATUS, "Device reply an bad status: " + cmd.getStatus()));
                return;
            }
            //Callback operation successful
        }

        @Override
        public void onErrCode(BluetoothDevice device, BaseError error) {
            //Callback operation failed
            //error - error information
        }
    });
