.. _ltv_format:

LTV Format
=======================================


LTV Format = Length(1Byte) + Type(1Byte) + Value(n Bytes)

=========== =============== ============
 Length       Type           Value
=========== =============== ============
  1 Byte      1 Byte          n Bytes 
 Length       Type            Content 
  N           1               N - 1
=========== =============== ============

**Field Parsing** 

* Length : Data length. The length from ``Type`` ~ ``Value``, i.e. (n + 1)
* Type : Function attribute.
* Value : Data content. Effective data length is (Length - 1)

.. _attr_bean:

Attribute Model
===========================================

Corresponding class name: **AttrBean** 

================ ================== ===================================
  Length(Bytes)      Name                 Notes
================ ================== ===================================
  1                size              Attribute length (length not including current Byte)
  1                attr type         Attribute type(:ref:`attr_fun_code`)
  size - 1         data              Attribute content
================ ================== ===================================

General Command Processing Example
----------------------------------------------------

.. code-block:: java

    //Step0: Get JL_BluetoothManager object
    JL_BluetoothManager manager = JL_BluetoothManager.getInstance(context);
    //Step1: Construct function command - e.g., get device information
    //mask = 0xffffffff;  -- Get all attributes
    CommandBase getDeviceInfoCmd = CommandBuilder.buildGetDeviceInfoCmd(mask);
    //Step2: Execute operation command and wait for result callback
    manager.sendCommandAsync(manager.getConnectedDevice(), getDeviceInfoCmd, manager.getBluetoothOption().getTimeoutMs(), new RcspCommandCallback() {
        //Callback reply command
        //Note: When no reply command, reply command itself
        @Override
        public void onCommandResponse(BluetoothDevice device, CommandBase cmd) {
            //Step3: Check device status
            if (cmd.getStatus() != StateCode.STATUS_SUCCESS) { //Device status abnormal, perform exception handling
                onErrCode(device, new BaseError(ErrorCode.SUB_ERR_RESPONSE_BAD_STATUS, "Device reply an bad status: " + cmd.getStatus()));
                return;
            }
            //Success callback
            //Step4: Check if there is reply data
            //Note - If command has no reply data, reply data is null
            boolean isHasResponse = cmd.getType() == CommandBase.FLAG_HAVE_PARAMETER_AND_RESPONSE
                    || cmd.getType() == CommandBase.FLAG_NO_PARAMETER_AND_RESPONSE;
            if (isHasResponse) {
                //Step5: Get corresponding command data
                GetTargetInfoCmd command = (GetTargetInfoCmd) cmd;
                //Step6: Get reply data
                TargetInfoResponse response = command.getResponse();
                if (null == response) { //Reply data is null, proving device reply data is problematic, perform exception handling
                    onErrCode(device, new BaseError(ErrorCode.SUB_ERR_DATA_FORMAT, "Response data is error."));
                    return;
                }
                //Process device reply data
            }
        }

        @Override
        public void onErrCode(BluetoothDevice device, BaseError error) {
            //Failure callback
            //error - error information
        }
    });


.. _receive_cmd_handle: 

Receive Command Processing Example
------------------------------------------------------

.. code-block:: java

    //Step0: Get JL_BluetoothManager object
    JL_BluetoothManager manager = JL_BluetoothManager.getInstance(context);
    //Step1: Add Bluetooth RCSP event callback
    manager.addEventListener(new BTRcspEventCallback() {
        //Callback device sent command
        @Override
        public void onDeviceCommand(BluetoothDevice device, CommandBase cmd) {
            super.onDeviceCommand(device, cmd);
            //Step2: Process command
            if (cmd.getId() != Command.CMD_ADV_DEV_REQUEST_OPERATION) {
                //Filter command
                return;
            }
            //Step3: Check if reply command is needed
            //Commands that need reply, need to reply result after processing
            //Commands that don't need reply, are complete after processing
            boolean isHasResponse = cmd.getType() == CommandBase.FLAG_HAVE_PARAMETER_AND_RESPONSE
                    || cmd.getType() == CommandBase.FLAG_NO_PARAMETER_AND_RESPONSE;
            //Only process target command - Request operation
            RequestAdvOpCmd requestAdvOpCmd = (RequestAdvOpCmd) cmd;
            //Step4: Get parameters
            RequestAdvOpParam param = requestAdvOpCmd.getParam();
            if (null == param) return; //Exception handling
            //Step5: Process data
            switch (param.getOp()) {
                case Constants.ADV_REQUEST_OP_UPDATE_CONFIGURE: //Update configuration
                    break;
                case Constants.ADV_REQUEST_OP_SYNC_TIME: //Synchronize time
                    break;
            }
            //Step6. Reply result
            if (isHasResponse) {
                //Assemble result
                requestAdvOpCmd.setStatus(StateCode.STATUS_SUCCESS);
                //Assemble parameters
                requestAdvOpCmd.setParam(null);
                //Send reply command
                manager.sendCommandResponse(device, requestAdvOpCmd);
            }
        }
    });
