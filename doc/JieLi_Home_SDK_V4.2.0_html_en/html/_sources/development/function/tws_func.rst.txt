    //Get RCSPController object

        @Override

            //Device broadcast info will be callback here
        }
    });
    //enable - Switch
    //Execute control device broadcast info function and wait for result callback
    controller.controlAdvBroadcast(controller.getUsingDevice(), enable, new OnRcspActionCallback<Boolean>() {
        @Override
        public void onSuccess(BluetoothDevice device, Boolean message) {
            //Success callback
            //enable = true, Enable device broadcast info, data will be in BTRcspEventCallback#onDeviceBroadcast callback
            //enable = false, Disable device broadcast info
        }

        @Override
        public void onError(BluetoothDevice device, BaseError error) {
            //Failure callback
            //error - Error info
        }
    });

============== ========================= ================================================================================================
 Function ID          Description                     Data Format
============== ========================= ================================================================================================
  0              Battery level            | left(1Byte)+right(1Byte)+charging bin(1Byte), not settable
                                          | Battery format: Bit7 -- charging status Bit6-0: battery level 
  1              Device name                name(32Bytes), ends with '\0'
  2              Button settings          | n * button settings
                                          | Button setting format: keyID(1Byte)+action(1Byte)+function(1Byte)
  3              Light settings           | n * light settings
                                          | Light setting format: scene(1Byte)+effect(1Byte)
  4              Microphone settings        mic_channel(1Byte)
  5              Work mode                  work_mode(1Byte)
  6              Product info               vid(2Bytes)+uid(2Bytes)+pid(2Bytes), not settable
  7              Access time                time(4Bytes), timestamp, unit is seconds
  8              In-ear detection           state(1Byte), 0 -- not enabled, 1 -- off, 2 -- on
  9              Language type              language(n bytes)
  10             ANC mode list            | modes(4Bytes), 32 bits, indicates mode list number
                                          | Bit0 -- off mode, Bit1 -- noise reduction mode, Bit2 -- transparency mode 
============== ========================= ================================================================================================


.. important::
    | 1. Parameters and data are :ref:`ltv_format` format 
    | 2. type indicates function identifier, explains data structure

Distinguish Neckband Headphones
========================================================


* When not connected, distinguish if TWS device is neckband headphones through broadcast packet

.. code-block:: java

    /** Is device type headset type(0x02,0x03) :
    0x00 -- Speaker type
    0x01 -- Charging bin type
    0x02 -- TWS headset type
    0x03 -- Headset type
    0x04--Sound card type
    0x05--Watch type */
    boolean isHeadset = UIHelper.isHeadsetByDeviceType(bleScanMessage.getDeviceType());
    //Is broadcast packet version neckband version
    boolean isNeck = isHeadset && bleScanMessage.getVersion() == SConstant.ADV_INFO_VERSION_NECK_HEADSET;
    return isNeck;

* When connected, distinguish if TWS device is neckband headphones through connection history record

.. code-block:: java

    /** Is chip type headset type(2,4):
    0:AI SDK (692X_AI smart speaker SDK)
    1:ST SDK (692X_standard speaker SDK)
    2:693x_TWS SDK
    3:695xSDK
    4:697x_TWS SDK
    5:696x_soundbox
    6:696x_tws
    7:695x_sound_card
    8:695x_sound_watch
    9:701x_sound_watch */
    boolean isHeadsetType = history != null && UIHelper.isHeadsetType(history.getChipType());
    //Is broadcast packet version neckband version
    boolean isNeck = isHeadsetType && history.getAdvVersion() == SConstant.ADV_INFO_VERSION_NECK_HEADSET;
    return isNeck;
